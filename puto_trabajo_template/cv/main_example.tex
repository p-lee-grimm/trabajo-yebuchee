%% Master CV - [YOUR_NAME]

%% Use as a master reference when tailoring role-specific CVs (/setup fills it in).
%%
%% Compile with: lualatex main_example.tex
%% (pdflatex often fails on modern MiKTeX with fontawesome5 font-expansion errors.)

\documentclass[11pt,a4paper,sans]{moderncv}
\moderncvstyle{banking}
\moderncvcolor{blue}

% Force the name and section headings to render in moderncv blue (color1).
% \namefont is the hook every name-style macro routes through, so this works on
% moderncv 2.3.1 (this server's version), which has no \firstnamestyle or
% \lastnamestyle -- renewing those raised "Command \firstnamestyle undefined",
% which nonstopmode swallowed, so the name never got the size or colour below.
% The name's size lives in \namefont: change 34/36 to restyle it.
\renewcommand*{\namefont}{\fontsize{34}{36}\bfseries\upshape}
\colorlet{firstnamecolor}{color1}
\colorlet{lastnamecolor}{color1}
\colorlet{namecolor}{color1}
\renewcommand*{\sectionstyle}[1]{{\sectionfont\color{color1}#1}}

\usepackage[utf8]{inputenc}
\usepackage{needspace}
% moderncv loads hyperref itself in an \AtEndPreamble hook, so \hypersetup must
% go in an \AtEndPreamble of our own -- a top-level \usepackage{hyperref} here
% raised "Option clash for package hyperref" on every build. pdfpagemode stays
% UseNone: this block runs after moderncv's own, so FullScreen would win and
% open the CV in fullscreen presentation mode.
\AtEndPreamble{\hypersetup{
    colorlinks=true,
    linkcolor=blue,
    filecolor=magenta,
    urlcolor=blue,
    pdftitle={[FIRST] [LAST] - CV},
    pdfpagemode=UseNone,
}}
\usepackage[scale=0.80]{geometry}
\usepackage{import}

% personal data
\name{[FIRST]}{[LAST]}
\address{[CITY, COUNTRY]}{}{}
\phone[mobile]{[PHONE]}
\email{[EMAIL]}
\extrainfo{\href{[LINKEDIN_URL]}{LinkedIn}}
